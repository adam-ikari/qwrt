# qwrt CLI 实现计划

日期：2026-08-15
依据：docs/superpowers/specs/2026-08-15-qwrt-cli-design.md（已批准）
执行方式：superpowers:subagent-driven-development 或 executing-plans

## Goal

新增 `qwrt` 可执行文件：以 WinterTC + W3C 风格直接运行 JS 脚本（单文件），脚本内可用全部 21 个 WinterTC Web API 模块；接口含 `qwrt script.js` / `qwrt -e 'code'` / 无参数 REPL / `--help` / `--version`；顶层跑完后等待 pending 异步任务（fetch/timer/streams）完成再退出；不引入任何 node 风格 API（无 process/Buffer/require）。

## Architecture

C 侧 CLI（src/cli.c）作为 qwrt 库的第一消费者：用 qwrt C API（qwrt_create/post_message/message_cb）驱动内部线程 + libuv loop，与 SDK 嵌入共享同一内核。脚本经 onmessage 命令通道 eval（复用 test_host.h 的 HostCtx 模式），console 输出经 message_cb/原生输出到进程 stdout/stderr。异步退出语义通过新增最小内核 API `qwrt_wait_idle()` 实现：qwrt 线程在消息排空 + 微任务 flush 后，检测 loop 无活跃 handle（除 wake async）即自动退出线程。

## Tech Stack

C99、QuickJS-ng、libuv（uv_a，经 qwrt_full）、mbedTLS/WAMR 等扩展（qwrt_full 内含）、GoogleTest（fork 子进程驱动 CLI）。

```c
/* qwrt.c 或 thread.c —— qwrt_wait_idle 实现（宿主线程调用，线程安全） */
void qwrt_wait_idle(qwrt_t *rt)
{
    if (!rt || rt->magic != QWRT_MAGIC) return;
    uv_mutex_lock(&rt->msg_mutex);
    rt->wait_idle = 1;
    uv_mutex_unlock(&rt->msg_mutex);
    uv_async_send(&rt->wake);
}

/* qwrt_internal.h 声明 walk 回调与 idle 判定 */
static void qwrt_idle_walk_cb(uv_handle_t *h, void *arg)
{
    /* arg = &(int busy)；wake async 的 handle->data 指向 rt，据此排除 */
    qwrt_t *rt = (qwrt_t *)h->data;
    int *busy = (int *)arg;
    if (h == (uv_handle_t *)&rt->wake) return;
    if (!uv_is_closing(h) && uv_is_active(h)) *busy = 1;
}

/* 除 wake async 外无活跃 handle 且消息队列空 → idle */
static int qwrt_loop_idle(qwrt_t *rt)
{
    int busy = 0;
    uv_mutex_lock(&rt->msg_mutex);
    int msgs = rt->msg_head != NULL;   /* 入站队列非空 */
    uv_mutex_unlock(&rt->msg_mutex);
    if (msgs) return 0;
    uv_walk(&rt->loop, qwrt_idle_walk_cb, &busy);
    return !busy;
}
```

thread.c 主循环改为：

```c

/*
 * qwrt CLI — standalone WinterTC runtime
 * 用法: qwrt [options] [script.js [args...]] | qwrt -e 'code'
 *       qwrt（无参数）→ REPL
 */
#include <qwrt/qwrt.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>

#define QWRT_CLI_VERSION "qwrt 0.1.0"

static void usage(FILE *out) {
    fprintf(out,
        "Usage: qwrt [options] [script.js [args...]]\n"
        "       qwrt -e 'code' [args...]\n"
        "       qwrt            (start REPL)\n"
        "\n"
        "Options:\n"
        "  -e, --eval <code>   evaluate <code> and exit\n"
        "  -h, --help          show this help\n"
        "  -v, --version       show version\n"
        "\n"
        "Runs a WinterTC-compatible JavaScript runtime (fetch, crypto,\n"
        "streams, timers, fs, ...). No Node.js APIs (process, require, ...).\n");
}

int main(int argc, char **argv) {
    /* 解析 -h/-v 与 -e；第一个非 flag 参数 = 脚本路径；其余 = 脚本参数 */
    for (int i = 1; i < argc; i++) {
        if (!strcmp(argv[i], "-h") || !strcmp(argv[i], "--help")) {
            usage(stdout);
            return 0;
        }
        if (!strcmp(argv[i], "-v") || !strcmp(argv[i], "--version")) {
            printf("%s\n", QWRT_CLI_VERSION);
            return 0;
        }
        if (!strcmp(argv[i], "-e") || !strcmp(argv[i], "--eval")) {
            if (i + 1 >= argc) { usage(stderr); return 2; }
            /* Task 5: eval 模式 */
        }
        /* 未知 flag */
        if (argv[i][0] == '-') { usage(stderr); return 2; }
        /* Task 4: 脚本模式 */
    }
    /* Task 6: REPL */
    return 0;
}
```

```c
/* 全局宿主状态 */
typedef struct {
    int done;                 /* eval 完成（含错误） */
    int exit_code;            /* 脚本异常 → 1 */
    char last_line[4096];     /* message_cb 输出缓冲（console 已分流，无需） */
} cli_host_t;

/* message_cb：跑在 qwrt 线程；CLI 收 eval 结果 */
static void cli_message_cb(qwrt_t *rt, const char *json, size_t len, void *data) {
    (void)rt;
    cli_host_t *h = (cli_host_t *)data;
    /* json: {"ok":true,"v":"..."} 或 {"ok":false,"e":"..."} */
    if (strstr(json, "\"ok\":false")) {
        fprintf(stderr, "%s\n", json);   /* 错误已含消息 */
        h->exit_code = 1;
    }
    h->done = 1;
}

/* WinterTC 桥 + onmessage 命令通道（对齐 test_host.h 的 kTestBootstrap）：
 * - globalThis.arguments: 脚本参数（不含 runtime parts，对齐 WinterCG
 *   proposal-cli-api 方向；不含可执行名与脚本路径）
 * - globalThis.env: 环境变量键值（最小形式）
 * - onmessage eval 命令通道：宿主 post {"cmd":"eval","code":...}
 *   → JS eval → postMessage({ok, v|e}) */
static const char *kCliBootstrap = R"JS(
globalThis.arguments = SCRIPT_ARGS_JSON;
globalThis.env = ENV_JSON;
globalThis.onmessage = function (e) {
  var d = e.data;
  if (d && d.cmd === 'eval') {
    try { postMessage({ok: true, v: JSON.stringify((0, eval)(d.code))}); }
    catch (err) { postMessage({ok: false, e: String(err)}); }
  }
};
)JS";
```

```c
/* C 字符串 → JSON 字符串字面量（转义反斜杠、引号、控制字符），返回 malloc 串 */
static char *json_escape(const char *s) {
    size_t n = strlen(s) * 6 + 1;
    char *out = malloc(n), *p = out;
    for (const unsigned char *c = (const unsigned char *)s; *c; c++) {
        switch (*c) {
        case '\\': *p++ = '\\'; *p++ = '\\'; break;
        case '"':  *p++ = '\\'; *p++ = '"'; break;
        case '\n': *p++ = '\\'; *p++ = 'n'; break;
        case '\r': *p++ = '\\'; *p++ = 'r'; break;
        case '\t': *p++ = '\\'; *p++ = 't'; break;
        default:
            if (*c < 0x20) { p += sprintf(p, "\\u%04x", *c); }
            else { *p++ = (char)*c; }
        }
    }
    *p = 0;
    return out;
}
```

**Files:**

- [ ] **Step 1: 实现 -e 模式**

把 Task 3 里 `-e` 分支的注释替换为实际逻辑：`code` 取自 `argv[i+1]`，后续参数为脚本参数；走与 Task 4 相同的 eval 通道 + wait_idle + 退出码逻辑。提取公共函数 `run_script(rt, code, args)` 避免重复（DRY）。

```c
/* 提取公共执行路径：建 runtime → eval code → 等结果 → wait_idle → destroy。
 * host 经 qwrt_set_runtime_data 绑定到 rt，cli_message_cb 用
 * qwrt_get_runtime_data(rt) 取回（可多次调用 run_code，host 不共享）。 */
static int run_code(const char *code, const char *const *args, int nargs) {
    cli_host_t host = {0};
    uv_mutex_init(&host.m); uv_cond_init(&host.c);

    char *bootstrap = build_bootstrap(args, nargs);   /* Task 7 注入 arguments/env */
    qwrt_config_t cfg = {};
    cfg.message_cb = cli_message_cb;
    cfg.host_data = &host;
    cfg.initial_script = bootstrap;
    qwrt_t *rt = qwrt_create(&cfg);
    free(bootstrap);
    if (!rt) { fprintf(stderr, "qwrt: runtime init failed\n"); return 1; }
    qwrt_set_runtime_data(rt, &host);

    char *cmd = malloc(strlen(code) * 2 + 64);
    sprintf(cmd, "{\"cmd\":\"eval\",\"code\":%s}", json_escape(code));
    qwrt_post_message(rt, cmd, strlen(cmd));
    free(cmd);

    uv_mutex_lock(&host.m);
    while (!host.done) uv_cond_wait(&host.c, &host.m);
    uv_mutex_unlock(&host.m);
    int exit_code = host.exit_code;

    qwrt_wait_idle(rt);      /* 等 pending 异步任务完成（fetch/timer） */
    qwrt_destroy(rt);
    uv_cond_destroy(&host.c); uv_mutex_destroy(&host.m);
    return exit_code;
}
```

（重构：`cli_message_cb` 从 `qwrt_get_runtime_data(rt)` 取 cli_host_t，而不是 config.host_data；这样 run_code 可多次调用。Task 4 的脚本模式同步重构。）

### Task 6: REPL

- [ ] **Step 1: 实现 REPL 循环**

无参数（且非 -e）时进入 REPL：打印 banner → 循环 `fgets` 读行 → eval（经 onmessage 通道）→ 打印 `v` 或 `e` → EOF（Ctrl-D）退出。

```c
static int repl_loop(qwrt_t *rt) {
    char line[8192];
    printf("qwrt %s (WinterTC runtime) — type JS, Ctrl-D to exit\n", QWRT_CLI_VERSION);
    fflush(stdout);
    while (fgets(line, sizeof line, stdin)) {
        /* 去除尾部换行 */
        line[strcspn(line, "\n")] = 0;
        if (!line[0]) continue;
        /* post eval → 同步等待 message_cb 回传结果 */
        cli_host_t *h = (cli_host_t *)qwrt_get_runtime_data(rt);
        uv_mutex_lock(&h->m);
        h->done = 0;
        uv_mutex_unlock(&h->m);
        char *cmd = malloc(strlen(line) * 2 + 64);
        sprintf(cmd, "{\"cmd\":\"eval\",\"code\":%s}", json_escape(line));
        qwrt_post_message(rt, cmd, strlen(cmd));
        free(cmd);
        uv_mutex_lock(&h->m);
        while (!h->done) uv_cond_wait(&h->c, &h->m);
        uv_mutex_unlock(&h->m);
        printf("%s\n", h->result);   /* {"ok":true,"v":...} 或 {"ok":false,"e":...} */
        fflush(stdout);
    }
    printf("\n");
    return 0;
}
```

注意：REPL 需要 message_cb 同步回传结果——`cli_host_t` 增加 `uv_mutex_t m; uv_cond_t c; int done; char result[4096];`；`cli_message_cb` 写 result 后 `uv_cond_signal(&h->c)`（与 test_host.h 的 HostCtx 模式一致）。Task 4 的脚本模式 `while (!host.done) usleep(1000)` 同步改为同一 mutex/cond 等待（去掉 usleep 轮询）。

```cpp
// test_cli_gtest.cpp — qwrt CLI 端到端（fork 子进程断言 stdout/stderr/退出码）
#include <gtest/gtest.h>
#include <cstdio>
#include <cstdlib>
#include <string>
#include <sys/wait.h>
#include <unistd.h>

#ifndef QWRT_CLI_BIN
#define QWRT_CLI_BIN "qwrt"
#endif

struct CliResult { int exit_code; std::string out, err; };

static CliResult run_cli(const std::vector<std::string>& args) {
    int out_pipe[2], err_pipe[2];
    if (pipe(out_pipe) != 0 || pipe(err_pipe) != 0) return {-1, "", ""};
    pid_t pid = fork();
    if (pid == 0) {
        /* child: 重定向 stdout/stderr 到管道，exec CLI */
        dup2(out_pipe[1], STDOUT_FILENO);
        dup2(err_pipe[1], STDERR_FILENO);
        close(out_pipe[0]); close(out_pipe[1]);
        close(err_pipe[0]); close(err_pipe[1]);
        std::vector<std::string> argv0 = { QWRT_CLI_BIN };
        argv0.insert(argv0.end(), args.begin(), args.end());
        std::vector<char*> cargv;
        for (auto& a : argv0) cargv.push_back(const_cast<char*>(a.c_str()));
        cargv.push_back(nullptr);
        execv(QWRT_CLI_BIN, cargv.data());
        _exit(127);   /* exec 失败 */
    }
    /* parent: 读管道 → waitpid */
    close(out_pipe[1]); close(err_pipe[1]);
    char buf[4096]; ssize_t n;
    std::string out, err;
    while ((n = read(out_pipe[0], buf, sizeof buf)) > 0) out.append(buf, n);
    while ((n = read(err_pipe[0], buf, sizeof buf)) > 0) err.append(buf, n);
    close(out_pipe[0]); close(err_pipe[0]);
    int status = 0;
    waitpid(pid, &status, 0);
    int code = WIFEXITED(status) ? WEXITSTATUS(status) : (WIFSIGNALED(status) ? 128 + WTERMSIG(status) : -1);
    return {code, out, err};
}
```

## 风险与注意

- **qwrt_wait_idle 语义**：设置后不得再 post_message；与 qwrt_destroy 二选一。wake async 的 data 指向 rt——idle 检测靠 `uv_walk` 排除 wake handle，需在 qwrt_internal.h 加 `int wait_idle` 字段。
- **console 分流是行为变更**：嵌入者若依赖 `[qwrt:N]` 前缀或全 stderr 输出会被影响；Task 1 独立 commit，可单独回滚。
- **REPL 同步等待**：必须用 mutex/cond（HostCtx 模式），不能简单 sleep；否则连续输入会串。
- **json_escape**：脚本代码注入 eval 命令 JSON 时需完整转义（引号/反斜杠/控制字符），复用 test_host.h 的 JSON_string 实现思路。
- **异步等待范围**：MVP 以 timer 类任务验证；网络 fetch 等待依赖本地服务，列入后续。
